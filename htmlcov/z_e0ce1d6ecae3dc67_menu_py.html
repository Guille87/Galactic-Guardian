40def _ajustar_texto(fuente, texto, ancho): 

41 """Parte `texto` en líneas que quepan en `ancho` píxeles (por palabras).""" 

42 lineas, actual = [], "" 

43 for palabra in texto.split(): 

44 prueba = f"{actual} {palabra}".strip() 

45 if actual and fuente.size(prueba)[0] > ancho: 

46 lineas.append(actual) 

47 actual = palabra 

48 else: 

49 actual = prueba 

50 if actual: 

51 lineas.append(actual) 

52 return lineas

60def _clamp_volumen(v): 

61 """Recorta a [0, 1] y redondea (arrastre libre). 

62 

63 Un valor fuera de [0, 1], aunque sea por 1e-17, hace que `pygame_gui` cree 

64 el slider fuera de rango y lo deje sin responder. 

65 """ 

66 try: 

67 v = float(v) 

68 except (TypeError, ValueError): 

69 return 0.5 

70 return round(min(1.0, max(0.0, v)), _DECIMALES_VOLUMEN)

73def _paso_volumen(v, subir): 

74 """Múltiplo de `_PASO_VOLUMEN` inmediatamente por encima/debajo de `v`. 

75 

76 0.27 → flecha arriba 0.3, flecha abajo 0.2. 0.20 → 0.3 / 0.1. 

77 """ 

78 n = round(v / _PASO_VOLUMEN, 6) 

79 objetivo = (math.floor(n) + 1 if subir else math.ceil(n) - 1) * _PASO_VOLUMEN 

80 return _clamp_volumen(objetivo)

191 def ejecutar(self): 

192 """Bucle principal del menú. Devuelve el siguiente estado 

193 ("JUGAR"/"JUGAR_SIN_FIN"/"SALIR").""" 

194 # Reinicio de estado por si volvemos desde una partida 

195 self.ejecutando = True 

196 self.estado = "PRINCIPAL" 

197 self.resultado = None 

198 self._sincronizar_estado() # por si Opciones se tocó desde la pausa 

199 if self._idioma_botones != i18n.idioma_actual(): # ídem el idioma 

200 self._crear_botones() 

201 

202 self._preparar_musica() # Solo activamos la música aquí, al lanzar el menú completo 

203 self.actualizaciones.comprobar_en_segundo_plano() # no-op si ya se lanzó 

204 

205 while self.ejecutando: 

206 time_delta = self.clock.tick(settings.FPS) / 1000.0 

207 

208 if self.estado == "PRINCIPAL": 

209 self._menu_principal() 

210 elif self.estado == "OPCIONES": 

211 self._menu_opciones(time_delta) 

212 elif self.estado == "PUNTUACIONES": 

213 self._menu_puntuaciones() 

214 elif self.estado == "MEJORAS": 

215 self._menu_mejoras() 

216 

217 return self.resultado or "SALIR"

409 def _inicializar_interfaz_opciones(self): 

410 """Crea el UIManager y los elementos de la interfaz **una sola vez**. 

411 

412 Las opciones se reparten en pestañas (Audio / Idioma / Controles): cada

414 Guardar y Volver son comunes a todas: actúan sobre lo tocado en cualquiera. 

415 

416 Antes se recreaba en cada entrada, dejando varios UIManager vivos.""" 

417 # Las etiquetas (Música/Efectos y el aviso de las flechas) se centran 

418 # por defecto en pygame_gui; con este tema quedan alineadas a la 

419 # izquierda, al ras de los sliders y los botones. 

420 tema = {"label": {"misc": {"text_horiz_alignment": "left"}}} 

421 self.ui_manager = pygame_gui.UIManager((settings.ANCHO, settings.ALTO), tema) 

422 self._idioma_ui = i18n.idioma_actual() 

423 self._elementos_pestana = {nombre: [] for nombre in PESTANAS} 

424

641 def _menu_opciones(self, time_delta): 

642 """Lógica de la pantalla de opciones usando pygame_gui.""" 

643 if self.ui_manager is None: # entrada directa sin pasar por _abrir_opciones 

644 self._abrir_opciones() 

645 elif self._idioma_ui != i18n.idioma_actual(): 

646 self._preparar_ui_opciones() # idioma cambiado: rehacer la UI con los textos nuevos 

647 

648 fondo = self.rm.get_image("imagen_fondo1") 

649 

650 # {botón de flecha: (destino, sube?)} 

651 flechas = { 

652 self.slider_musica.left_button: ("musica", False), 

653 self.slider_musica.right_button: ("musica", True), 

654 self.slider_efectos.left_button: ("efectos", False), 

655 self.slider_efectos.right_button: ("efectos", True), 

656 } 

657 sliders = {self.slider_musica: "musica", self.slider_efectos: "efectos"} 

658 

659 for event in pygame.event.get(): 

660 if event.type == pygame.QUIT: 

661 self._deshacer_cambios() 

662 self.ejecutando = False 

663 self.estado = "PRINCIPAL" 

664 self.resultado = "SALIR" 

665 

666 # --- Reasignación de teclas: la siguiente pulsación es la respuesta --- 

667 elif event.type == pygame.KEYDOWN and self._reasignando_accion is not None: 

668 self._procesar_tecla_reasignada(event.key) 

669 

670 # --- Eventos de pygame_gui (API 0.6+: cada evento con su propio type) --- 

671 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element in flechas: 

672 # Solo marcamos la dirección; el ajuste se hace al final del frame. 

673 self._flecha_pendiente = flechas[event.ui_element] 

674 

675 elif event.type == pygame_gui.UI_HORIZONTAL_SLIDER_MOVED and event.ui_element in sliders: 

676 destino = sliders[event.ui_element] 

677 if self._ignorar_moved == destino: 

678 # MOVED tardío de una flecha que ya cuadramos el frame anterior. 

679 self._ignorar_moved = None 

680 event.ui_element.set_current_value( 

681 self.vol_musica if destino == "musica" else self.vol_efectos) 

682 elif self._flecha_pendiente is None or self._flecha_pendiente[0] != destino: 

683 # Arrastre de la barra: valor libre, solo recortado. 

684 self._fijar_volumen(destino, _clamp_volumen(event.ui_element.get_current_value())) 

685 

686 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element in self._botones_pestana: 

687 self._mostrar_pestana(self._botones_pestana[event.ui_element]) 

688 

689 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element in self._botones_idioma: 

690 self._cambiar_idioma(self._botones_idioma[event.ui_element]) 

691 

692 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element == self.btn_temblor: 

693 preferencias.establecer_temblor(not preferencias.temblor_activado()) 

694 self._actualizar_boton_temblor() 

695 

696 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element in self._acciones_por_boton: 

697 self._empezar_reasignacion(self._acciones_por_boton[event.ui_element]) 

698 

699 elif event.type == pygame_gui.UI_BUTTON_PRESSED and event.ui_element == self.btn_restaurar_controles: 

700 self.controles = dict(controles.POR_DEFECTO) 

701 for accion in controles.ACCIONES: 

702 self._actualizar_texto_control(accion) 

703 

704 elif event.type == pygame_gui.UI_BUTTON_PRESSED: 

705 if event.ui_element == self.btn_guardar: 

706 config.guardar_configuracion(self.vol_musica, self.vol_efectos, self.controles, 

707 idioma=i18n.idioma_actual(), 

708 temblor=preferencias.temblor_activado()) 

709 self.estado = "PRINCIPAL" 

710 elif event.ui_element == self.btn_volver: 

711 self._deshacer_cambios() 

712 self.estado = "PRINCIPAL" 

713 

714 self.ui_manager.process_events(event) 

715 

716 # Flecha ◄ ►: pygame_gui ya ha aplicado su propio ±0.1 en este mismo 

717 # frame. Lo corregimos AQUÍ (antes de dibujar) al múltiplo de 0.1 

718 # anterior/siguiente al valor que había, para que no se vea el salto 

719 # intermedio; el MOVED que pygame_gui emite se ignora el próximo frame. 

720 if self._flecha_pendiente is not None: 

721 destino, sube = self._flecha_pendiente 

722 self._flecha_pendiente = None 

723 actual = self.vol_musica if destino == "musica" else self.vol_efectos 

724 self._fijar_volumen(destino, _paso_volumen(actual, sube), feedback_reiniciar=True) 

725 self._ignorar_moved = destino 

726 

727 self.ui_manager.update(time_delta) 

728 

729 # Al MANTENER pulsada una flecha >0.2 s, pygame_gui arranca un 

730 # desplazamiento continuo y rápido del slider. Reseteamos su acumulador 

731 # mientras la flecha siga pulsada para que solo cuente el clic limpio. 

732 for sl in (self.slider_musica, self.slider_efectos): 

733 if sl.left_button.held or sl.right_button.held: 

734 sl.button_held_repeat_acc = 0.0 

735 self.pantalla.blit(fondo, (0, 0)) 

736 self.pantalla.blit(self._overlay_oscuro(), (0, 0)) 

737 

738 # Renderizar textos (Título, etiquetas de sliders) 

739 txt_opciones = self.font_titulo.render(t("opciones.titulo"), True, (255, 255, 255)) 

740 self.pantalla.blit(txt_opciones, (50, 50)) 

741 

742 if (self._pestana == "controles" and self._aviso_conflicto 

743 and time.time() < self._aviso_conflicto_hasta): 

744 aviso = self.font_version.render(self._aviso_conflicto, True, (255, 120, 120)) 

745 self.pantalla.blit(aviso, (50, 497)) 

746 

747 self.ui_manager.draw_ui(self.pantalla) 

748 pygame.display.flip()

750 def mostrar_solo_opciones(self): 

751 """Muestra las opciones y retorna el control cuando se pulsa Guardar o Volver. 

752 

753 Se usa desde la pausa de la partida. Devuelve "SALIR" si el usuario cerró 

754 la ventana (para que el motor propague el cierre), o None en caso normal. 

755 """ 

756 self._abrir_opciones() 

757 

758 bucle_opciones = True 

759 while bucle_opciones: 

760 time_delta = self.clock.tick(settings.FPS) / 1000.0 

761 self._menu_opciones(time_delta) 

762 

763 # Salida normal: Guardar o Volver dejan el estado en PRINCIPAL 

764 if self.estado == "PRINCIPAL": 

765 bucle_opciones = False 

766 

767 # Cierre de ventana durante la pausa: re-emitimos QUIT y salimos 

768 if self.resultado == "SALIR": 

769 pygame.event.post(pygame.event.Event(pygame.QUIT)) 

770 return "SALIR" 

771 

772 return None
