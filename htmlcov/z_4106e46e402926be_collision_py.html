6class CollisionManager: 

7 """Detección de colisiones y su respuesta **mecánica**. 

8 

9 Colaboradores explícitos (mecánica pura): los grupos de entidades, el 

10 jugador, los efectos, el audio y el registro de cooldown de contacto. 

11 

12 Las consecuencias de **reglas de partida** (puntuación, loot, progresión de 

13 nivel, muerte/reaparición del jugador) NO se aplican aquí: se delegan en 

14 `reglas`, que debe ofrecer dos métodos: 

15 

16 - ``reglas.al_eliminar_enemigo(enemigo)`` — puntuación, loot, transición 

17 de jefe. Se llama **después** de que este manager haya hecho la parte 

18 mecánica (``kill``, explosión, purga del cooldown). 

19 - ``reglas.manejar_impacto_jugador()`` — destello / explosión / pérdida de 

20 vida y reaparición. Se llama tras aplicar el daño al jugador. 

21 

22 Colisiones: 

23 - Balas <-> naves: colisión circular (``collide_circle``, usa ``radius``). 

24 - Contacto cuerpo a cuerpo y recogida de ítems: colisión rectangular. 

25 """ 

26 

27 def __init__(self, entity_manager, jugador, effect_manager, audio_manager, 

28 enemigos_golpeados, reglas): 

29 self.em = entity_manager 

30 self.jugador = jugador 

31 self.efectos = effect_manager 

32 self.audio = audio_manager 

33 self.enemigos_golpeados = enemigos_golpeados 

34 self.reglas = reglas

36 def actualizar(self, tiempo_juego): 

37 self._balas_jugador_vs_enemigos() 

38 self._balas_enemigo_vs_jugador() 

39 self._contacto_cuerpo_a_cuerpo(tiempo_juego) 

40 self._jugador_vs_items()

43 def _balas_jugador_vs_enemigos(self): 

44 impactos = pygame.sprite.groupcollide( 

45 self.em.balas, self.em.enemigos, True, False, 

46 collided=pygame.sprite.collide_circle, 

47 ) 

48 for bala, enemigos_tocados in impactos.items(): 

49 enemigo = enemigos_tocados[0] # una bala daña a un solo enemigo 

50 enemigo.take_damage(bala.danio) 

51 if enemigo.salud <= 0: 

52 self._eliminar_enemigo(enemigo) 

53 

54 # Un solo "golpe" por frame, no uno por bala (evita saturar el mixer) 

55 if impactos: 

56 self.audio.reproducir_efecto("golpe") 

57 

58 def _balas_enemigo_vs_jugador(self): 

59 tocadas = pygame.sprite.spritecollide( 

60 self.jugador, self.em.balas_enemigo, True, 

61 collided=pygame.sprite.collide_circle, 

62 ) 

63 for bala in tocadas: 

64 self.jugador.recibir_danio(bala.danio) 

65 self.reglas.manejar_impacto_jugador() 

66 

67 if tocadas: 

68 self.audio.reproducir_efecto("golpe") 

69 

70 def _contacto_cuerpo_a_cuerpo(self, ahora): 

71 for enemigo in pygame.sprite.spritecollide(self.jugador, self.em.enemigos, False): 

72 ultimo = self.enemigos_golpeados.get(enemigo, 0) 

73 if ahora - ultimo >= settings.CONTACTO_COOLDOWN_MS: 

74 self.jugador.recibir_danio(settings.DANIO_CONTACTO) 

75 self.audio.reproducir_efecto("golpe") 

76 self.enemigos_golpeados[enemigo] = ahora 

77 self.reglas.manejar_impacto_jugador() 

78 enemigo.salud -= 1 

79 

80 if enemigo.salud <= 0: 

81 self._eliminar_enemigo(enemigo) 

82 

83 def _jugador_vs_items(self): 

84 for item in pygame.sprite.spritecollide(self.jugador, self.em.items, True): 

85 self.audio.reproducir_efecto("item") 

86 item.aplicar_efecto(self.jugador)

89 def _eliminar_enemigo(self, enemigo): 

90 if enemigo not in self.em.enemigos: 

91 return # ya procesado en este mismo frame (p. ej. dos balas a la vez) 

92 

93 enemigo.kill() 

94 self.enemigos_golpeados.pop(enemigo, None) 

95 self.efectos.crear_explosion(enemigo.rect.center)

97 # Puntuación, loot y transición de jefe: reglas de partida. 

98 self.reglas.al_eliminar_enemigo(enemigo)
