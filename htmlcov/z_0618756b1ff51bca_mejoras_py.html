1"""Mejoras permanentes de la nave (solo datos) y el `Bonus` que suman. 

2 

3Se compran con monedas en la pantalla "Mejoras" del menú (ver `progresion.py`). 

4Forman tres ramas; dentro de una rama cada mejora se desbloquea al comprar la 

5anterior (`requiere`). Añadir o retocar una es editar `MEJORAS`: las ramas pueden 

6tener los nodos que hagan falta (la pantalla se desplaza). Sus textos están en el 

7catálogo (`mejoras.<id>.nombre` y `mejoras.<id>.desc`) y los de las ramas en 

8`mejoras.rama_<rama>`. 

9 

10Un `efecto` es un diccionario campo del `Bonus` -> valor. Los bonus suben el 

11valor **de arranque** de la nave, nunca los topes de `Jugador.CONFIG` (velocidad 6, 

128 disparos por segundo, 3 balas, 30 de daño). Los costes y los valores salen del 

13modelo de `tools/balance.py`: si cambias uno, vuelve a correrlo. 

14""" 

15 

16from dataclasses import dataclass, field, fields 

17 

18RAMAS = ("ataque", "defensa", "utilidad") 

19 

20 

21@dataclass(frozen=True) 

22class Bonus: 

23 """Suma de los efectos de las mejoras compradas (todo a cero = sin mejoras).""" 

24 salud_extra: int = 0 # salud máxima inicial (en puntos de salud reales) 

25 vidas_extra: int = 0 # vidas iniciales 

26 danio_extra: int = 0 # daño inicial (por bala, en puntos reales) 

27 disparos_extra: float = 0.0 # disparos por segundo de más 

28 balas_extra: int = 0 # balas por disparo de más (2 = doble, 3 = triple) 

29 velocidad_extra: float = 0.0 # velocidad inicial 

30 regen_s: float = 0.0 # salud que se recupera por segundo tras unos segundos sin daño 

31 invulnerable_extra_ms: int = 0 # escudo de reaparición más largo 

32 monedas_pct: float = 0.0 # +% de monedas (0.25 = +25 %) 

33 combo_factor: float = 1.0 # multiplica los umbrales del combo (<1 = más fácil) 

34 

35 

36@dataclass(frozen=True) 

37class Mejora:

42 requiere: str = None # id de la mejora anterior de su rama 

43 icono: str = None # nombre de imagen (`config.RECURSOS`) para su nodo, o ninguno

82POR_ID = {m.id: m for m in MEJORAS} 

83

85def de_la_rama(rama): 

86 """Las mejoras de una rama, en el orden en que se desbloquean.""" 

87 return [m for m in MEJORAS if m.rama == rama] 

88 

89 

90def calcular_bonus(ids): 

91 """`Bonus` que suman las mejoras `ids` (ids desconocidos se ignoran).""" 

92 valores = {f.name: f.default for f in fields(Bonus)} 

93 for id_ in ids: 

94 mejora = POR_ID.get(id_) 

95 if mejora is None: 

96 continue 

97 for campo, valor in mejora.efecto.items(): 

98 if campo == "combo_factor": # se multiplican 

99 valores[campo] *= valor 

100 else: # el resto se suma 

101 valores[campo] += valor 

102 return Bonus(**valores)
