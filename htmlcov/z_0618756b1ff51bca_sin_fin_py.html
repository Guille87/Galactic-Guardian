1"""Curva de dificultad del modo sin fin (solo datos y funciones puras). 

2 

3La partida se divide en **oleadas** numeradas desde 1. Cada una es una 

4`DefinicionNivel` (la misma estructura que un nivel de campaña, así que 

5`WaveManager` la lee igual) generada por `definicion_oleada(n)`: 

6 

7- Una oleada normal dura `OLEADA_MS` y no tiene jefe; cuando se acaba el tiempo, 

8 `Juego` pasa a la siguiente. 

9- Cada `JEFE_CADA` oleadas toca una **de jefe**: el jefe sale nada más empezar 

10 (tras una breve espera con la música cambiada), no aparece ningún otro enemigo 

11 mientras dura, y la oleada acaba cuando cae. 

12 

13No hay techo: el intervalo entre apariciones se estabiliza en `INTERVALO_MIN`, 

14pero la vida de los enemigos y de los jefes sigue creciendo con el número de 

15oleada (tablas de `escalado.py`, que siguen más allá del nivel 5) y la puntuación se multiplica por él. 

16""" 

17 

18import math 

19from functools import lru_cache 

20 

21from src.core.niveles import DefinicionNivel, Fase, T1, T2, T3 

22 

23OLEADA_MS = 45000 # duración de una oleada normal 

24JEFE_CADA = 5 # una de cada N oleadas es de jefe 

25ESPERA_JEFE_MS = 3000 # margen entre el cambio de música y la aparición del jefe 

26 

27INTERVALO_INICIAL = (900, 1100) # (mín, máx) ms entre apariciones en la oleada 1 (como el nivel 1) 

28INTERVALO_PASO_MS = 100 # cuánto baja cada oleada, en cada extremo 

29INTERVALO_MIN = (400, 600) # suelo: el ritmo del último nivel de la campaña 

30 

31 

32def es_oleada_de_jefe(oleada): 

33 return oleada % JEFE_CADA == 0 

34 

35 

36def intervalo_spawn(oleada): 

37 """(mín, máx) ms entre apariciones: baja `INTERVALO_PASO_MS` por oleada hasta el suelo.""" 

38 bajada = INTERVALO_PASO_MS * (max(oleada, 1) - 1) 

39 return (max(INTERVALO_MIN[0], INTERVALO_INICIAL[0] - bajada), 

40 max(INTERVALO_MIN[1], INTERVALO_INICIAL[1] - bajada)) 

41 

42 

43def enemigos(oleada): 

44 """Reparto de enemigos normales (uniforme; repetir una clase le da más peso).""" 

45 if oleada <= 1: 

46 return (T1,) 

47 if oleada == 2: 

48 return (T1, T1, T2) 

49 if oleada <= 4: 

50 return (T1, T2, T3) 

51 return (T1, T2, T2, T3, T3) 

52 

53 

54def definicion_oleada(oleada): 

55 """`DefinicionNivel` de la oleada `oleada` (1, 2, ...; menos de 1 cuenta como 1).""" 

56 return _definicion_oleada(max(oleada, 1)) 

57 

58 

59@lru_cache(maxsize=None) 

60def _definicion_oleada(oleada): 

61 if es_oleada_de_jefe(oleada): 

62 # El jefe llega ya (tiempo 0 de la oleada): no salen enemigos normales. 

63 tiempo_jefe_ms = 0 

64 else: 

65 tiempo_jefe_ms = math.inf # nunca: la oleada termina por tiempo, en `Juego` 

66 return DefinicionNivel( 

67 fases=(Fase(0, enemigos(oleada)),), 

68 tiempo_jefe_ms=tiempo_jefe_ms, 

69 espera_jefe_ms=ESPERA_JEFE_MS, 

70 intervalo_spawn=intervalo_spawn(oleada), 

71 )
