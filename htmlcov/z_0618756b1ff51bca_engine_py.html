184 def actualizar(self, dt): 

185 """Actualiza el estado del juego. `dt` en segundos.""" 

186 if self.jugador.vidas <= 0: 

187 self.juego_terminado() 

188 return

193 # 1. Entradas y Generación 

194 teclas = pygame.key.get_pressed() 

195 self.jugador.mover(teclas, self.pantalla, dt)

199 self.entity_manager.actualizar(dt)

213 def _gestionar_generacion_enemigos(self): 

214 """Maneja el timing para spawnear enemigos mediante el WaveManager.""" 

215 ahora = self.tiempo_juego 

216 if ahora > self.tiempo_proximo_enemigo: 

217 nuevo = self.wave_manager.spawn_enemigo( 

218 ahora - self.inicio_juego, ahora, self.jugador, self.nivel 

219 ) 

220 if nuevo: 

221 self.entity_manager.agregar_enemigo(nuevo) 

222 

223 if isinstance(nuevo, Jefe): 

224 self.jefe = nuevo 

225 

226 self.tiempo_proximo_enemigo = ahora + random.randint(self.MIN_TIEMPO_GENERACION, self.MAX_TIEMPO_GENERACION)
