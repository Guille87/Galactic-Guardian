399 def _actualizar_transicion_nivel(self, dt): 

400 """Fotograma a fotograma de la secuencia: centrar -> subir -> esperar. 

401 

402 Valores ajustables en `settings.py`: `TRANSICION_VEL_LATERAL` (rapidez 

403 al centrarse), `TRANSICION_VEL_SUBIDA` (rapidez al subir), 

404 `TRANSICION_FONDO_ACELERACION` / `TRANSICION_FONDO_RAMPA_MS` (cuánto y 

405 en cuánto tiempo se acelera el fondo) y `TRANSICION_ESPERA_MS` (pausa 

406 tras desaparecer la nave, antes de la pantalla). 

407 """ 

408 factor = dt * settings.FPS 

409 jugador = self.jugador 

410 centro_x = self.pantalla_ancho // 2 

411 

412 if self.transicion_fase == "centrar": 

413 paso = settings.TRANSICION_VEL_LATERAL * factor 

414 diferencia = centro_x - jugador.rect.centerx 

415 if abs(diferencia) <= paso: 

416 jugador.rect.centerx = centro_x 

417 self.transicion_fase = "subir" 

418 else: 

419 jugador.rect.centerx += paso if diferencia > 0 else -paso 

420 

421 elif self.transicion_fase == "subir": 

422 jugador.rect.y -= settings.TRANSICION_VEL_SUBIDA * factor 

423 

424 # El fondo acelera de forma gradual mientras la nave sube. 

425 rampa = min(1.0, self.transicion_tiempo_fase / settings.TRANSICION_FONDO_RAMPA_MS) 

426 self.background.velocidad = settings.FONDO_VELOCIDAD_NORMAL * ( 

427 1 + rampa * (settings.TRANSICION_FONDO_ACELERACION - 1) 

428 ) 

429 self.transicion_tiempo_fase += dt * 1000 

430 

431 if jugador.rect.bottom < -settings.TRANSICION_MARGEN_SALIDA: # ha desaparecido por arriba 

432 self.transicion_fase = "espera" 

433 self.transicion_tiempo_fase = 0.0 

434 

435 elif self.transicion_fase == "espera": 

436 self.transicion_tiempo_fase += dt * 1000 

437 if self.transicion_tiempo_fase >= settings.TRANSICION_ESPERA_MS: 

438 self._finalizar_transicion_fin_de_nivel() 

439 return 

440 

441 self.background.update(dt)

462 """`destino`: "game_over" o "victoria". Si la puntuación entra en el 

463 top 10 se pide el nombre primero (`pidiendo_nombre_para` recuerda a qué 

464 pantalla ir después); si no, se muestra esa pantalla directamente."""

475 def _cobrar_monedas(self): 

476 """Ingresa en la progresión las monedas que aún no se han cobrado de esta 

477 partida (`MONEDAS_PUNTOS` puntos por moneda, más el % de la mejora 

478 "Botín"). Es idempotente: llamarla varias veces no duplica nada, así que 

479 se llama en cada forma de terminar o pausar el progreso de una partida.""" 

480 if self.progresion is None: 

481 return 

482 total = int((self.puntuacion // settings.MONEDAS_PUNTOS) * (1 + self.bonus.monedas_pct)) 

483 nuevas = total - self.monedas_cobradas 

484 if nuevas > 0: 

485 self.progresion.ingresar(nuevas) 

486 self.monedas_cobradas = total

509 """Abre la pantalla de opciones sobre la pausa de la partida."""

525 def ejecutar(self): 

526 """Ejecuta el bucle de la partida. Devuelve el siguiente estado ("MENU"/"SALIR").""" 

527 self.ejecutando = True 

528 

529 while self.ejecutando: 

530 pygame.display.set_caption(f"Galactic Guardian v{__version__}") 

531 

532 # Si manejar_eventos() devuelve False, salimos del bucle 

533 if not self.input_handler.manejar_eventos(): 

534 self.ejecutando = False 

535 continue 

536 

537 # Si el juego está pausado, solo dibujar la pantalla y continuar al siguiente ciclo 

538 if self.pausado: 

539 self.dibujar() 

540 self.reloj.tick(settings.FPS) 

541 continue 

542 

543 # dt en segundos, acotado para evitar saltos tras un parón (breakpoint, 

544 # arrastre de ventana, primer frame...) 

545 dt = min(self.reloj.tick(settings.FPS) / 1000.0, 3.0 / settings.FPS) 

546 

547 self.actualizar(dt) 

548 self.dibujar() 

549 

550 # Limpieza única de audio al abandonar la partida 

551 self._cobrar_monedas() # por si se salió por un camino que no la cobró (idempotente) 

552 self.audio_manager.detener_toda_la_musica() 

553 return self.resultado
