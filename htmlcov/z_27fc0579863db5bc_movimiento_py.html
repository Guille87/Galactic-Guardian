1"""Utilidad de movimiento independiente de los FPS. 

2 

3El `rect` de pygame trabaja con enteros, así que un desplazamiento fraccionario 

4por frame (p. ej. 2.4 px) perdería la parte decimal en cada paso. `MovimientoSubpixel` 

5acumula ese resto y lo aplica cuando suma un píxel entero. 

6 

7El `rect` sigue siendo la fuente de verdad: si otro código lo reposiciona 

8directamente (reaparición del jugador, rebote del jefe...), el siguiente 

9desplazamiento parte de esa nueva posición sin desincronizarse. 

10""" 

11 

12import math 

13 

14import pygame 

15 

16from src.core import settings 

17 

18 

19class MovimientoSubpixel: 

20 def _init_subpixel(self): 

21 self._resto = pygame.Vector2() 

22 # Píxeles realmente movidos en el último frame (para el HUD de debug). 

23 self._ultimo_desplazamiento = pygame.Vector2() 

24 

25 def _desplazar(self, vx, vy, dt): 

26 """Desplaza el rect según una velocidad en px/frame-a-60fps y el dt real.""" 

27 factor = dt * settings.FPS 

28 dx = vx * factor + self._resto.x 

29 dy = vy * factor + self._resto.y 

30 

31 # Redondeo al entero más cercano (no truncado hacia cero): el resto queda 

32 # en [-0.5, 0.5), así que el rect nunca se aleja más de medio píxel de su 

33 # posición ideal. Con int() podía quedarse hasta 1 px por detrás y en un 

34 # ángulo poco inclinado eso se veía como escalones / "la bala se estanca". 

35 ix = math.floor(dx + 0.5) 

36 iy = math.floor(dy + 0.5) 

37 self.rect.x += ix 

38 self.rect.y += iy 

39 

40 self._resto.x = dx - ix 

41 self._resto.y = dy - iy 

42 self._ultimo_desplazamiento.update(ix, iy)
