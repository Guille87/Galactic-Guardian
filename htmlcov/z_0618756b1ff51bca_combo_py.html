1"""Combo de puntuación (lógica pura, sin pygame). 

2 

3La **racha** cuenta las bajas seguidas sin recibir daño; el **multiplicador** de 

4puntuación sale de ella según `settings.COMBO_UMBRALES` (×1 al principio, +1 por 

5cada umbral alcanzado). Solo se rompe al recibir un golpe (`romper`): no caduca 

6por tiempo. 

7""" 

8 

9from src.core import settings 

10 

11 

12class Combo: 

13 def __init__(self, factor=1.0): 

14 """`factor` multiplica los umbrales de `settings.COMBO_UMBRALES` (una mejora 

15 permanente lo baja: `<1` = el combo sube más fácil).""" 

16 self.racha = 0 

17 self.umbrales = tuple(max(1, round(u * factor)) for u in settings.COMBO_UMBRALES) 

18 

19 @property 

20 def multiplicador(self): 

21 return 1 + sum(1 for umbral in self.umbrales if self.racha >= umbral) 

22 

23 @property 

24 def es_maximo(self): 

25 return self.multiplicador == 1 + len(self.umbrales) 

26 

27 def progreso(self): 

28 """Fracción (0..1) que falta recorrer hasta el siguiente escalón; 1.0 en el máximo.""" 

29 if self.es_maximo: 

30 return 1.0 

31 nivel = self.multiplicador - 1 # escalones ya superados 

32 desde = self.umbrales[nivel - 1] if nivel else 0 

33 hasta = self.umbrales[nivel] 

34 return (self.racha - desde) / (hasta - desde) 

35 

36 def sumar_baja(self): 

37 self.racha += 1 

38 

39 def romper(self): 

40 self.racha = 0
